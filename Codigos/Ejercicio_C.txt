#define F_CPU 16000000UL // Frecuencia reloj del ATmega2560 (16 MHz)
#include <avr/io.h>
#include <util/delay.h>
#include <avr/interrupt.h>
#include <stdio.h>
#include <stdlib.h>
#include <math.h>


// Configuración de la Matiz

#define WS2812_PIN PH3 // Pin datos matriz (6)
#define WS2812_DDR DDRH  // Registro dirección Puerto H
#define WS2812_PORT PORTH  // Registro salida Puerto H

#define NUM_LEDS 64 // Total leds a usar en matriz 8x8

// Estructura color RGB en orden GRB (formato interno WS2812)
typedef struct {
	uint8_t g;
	uint8_t r;
	uint8_t b;
} cRGB;

cRGB leds[NUM_LEDS]; // Búfer memoria de 64 LEDs

// Envío bit a bit con temporización estricta por ensamblador inline (a 16 MHz)
void ws2812_send_byte(uint8_t b) {
	for (uint8_t i = 0; i < 8; i++) {
		if (b & 0x80) {
			// Bit 1 (Pulso alto 700ns, bajo 600ns)
			WS2812_PORT |= (1 << WS2812_PIN); // se enciende el pin
			asm volatile ("nop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\t"); // Se queda congelado (nanosegundos)
			WS2812_PORT &= ~(1 << WS2812_PIN); // Se apaga el pin
			asm volatile ("nop\n\tnop\n\t"); // Mini espera antes de enviar siguente señal
			} else {
			// Bit 0 ( Pulso alto 350ns, bajo 800ns)
			WS2812_PORT |= (1 << WS2812_PIN);
			asm volatile ("nop\n\tnop\n\t");
			WS2812_PORT &= ~(1 << WS2812_PIN);
			asm volatile ("nop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\t");
		}
		b <<= 1;
	}
}

// Envía trama completa de búfer hacia matriz, desactiva interrupciones
void ws2812_show(void) {
	cli(); // Desactiva interrupciones para no alterar tiempos
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		ws2812_send_byte(leds[i].g);
		ws2812_send_byte(leds[i].r);
		ws2812_send_byte(leds[i].b);
	}
	sei(); // Reactivar interrupciones
	_delay_us(60); // Senal Reset, fija colores
}

// Apaga 64 LEDs de la matriz
void ws2812_clear(void) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		leds[i].r = 0;
		leds[i].g = 0;
		leds[i].b = 0;
	}
	ws2812_show();
}

// Muestreo
// Enciende LEDs 0, 1 y 2 para iluminación RGB
void ws2812_set_eval_leds(uint8_t r, uint8_t g, uint8_t b) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		if (i < 3) {
			leds[i].r = r;
			leds[i].g = g;
			leds[i].b = b;
			} else { // Si no son los preimeros tres leds s epongan en 0
			leds[i].r = 0;
			leds[i].g = 0;
			leds[i].b = 0;
		}
	}
	ws2812_show(); // se llama para enviar información recibida
}

// Resultado
// Enciende LEDs 8 al 63
void ws2812_set_display_leds(uint8_t r, uint8_t g, uint8_t b) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		if (i < 8) {
			// LEDs 0 al 7 apagada
			leds[i].r = 0;
			leds[i].g = 0;
			leds[i].b = 0;
			} else {
			// LEDs 8 al 63 encendidos con color clasificado
			leds[i].r = r;
			leds[i].g = g;
			leds[i].b = b;
		}
	}
	ws2812_show(); // Refrescaa la matriz
}

// Servomotor ( Pin 11, PWM)

void Servo_init(void) {
	DDRB |= (1 << PB5); // Pin 11 como salida
	
	// Modo Fast PWM con TOP en ICR1 (Modo 14)
	TCCR1A = (1 << COM1A1) | (1 << WGM11);
	TCCR1B = (1 << WGM13) | (1 << WGM12) | (1 << CS11); // Prescaler de 8
	
	ICR1 = 39999; // TOP para frecuencia de PWM a 50 Hz (período de 20 ms)
}

// Mapea y mueve el servomotor en el rango de 0° a 90°
void Servo_set_angle(uint8_t angle) {
	if (angle > 90) angle = 90; // Límite de seguridad
	
	// Convierte grados (0° a 90°) a ancho de pulso PWM en ticks del Timer 1
	uint16_t pulse = 2000 + ((uint32_t)angle * 2000) / 90;
	OCR1A = pulse;
}


// Comunicación UART 0 (9600 Baudios)
void UART0_init(unsigned int ubrr) {
	UBRR0H = (unsigned char)(ubrr >> 8);
	UBRR0L = (unsigned char)ubrr;
	UCSR0B = (1 << TXEN0); // Habilitar Transmisión TX
	UCSR0C = (1 << UCSZ01) | (1 << UCSZ00); // 8 bits de datos, 1 bit de parada
}

void UART0_send_string(const char* str) {
	while (*str) {
		while (!(UCSR0A & (1 << UDRE0))); // Esperar búfer libre
		UDR0 = *str++;
	}
}

// Convertidor  análogo - digital (ADC A0)
void ADC_init(void) {
	ADMUX = (1 << REFS0); // Tensión referencia (5V)
	ADCSRA = (1 << ADEN) | (1 << ADPS2) | (1 << ADPS1) | (1 << ADPS0); // Prescaler de 128
}

uint16_t ADC_read(uint8_t channel) {
	ADCSRB &= ~(1 << MUX5);
	ADMUX = (ADMUX & 0xF0) | (channel & 0x07); // Seleccionar canal ADC (A0)
	ADCSRA |= (1 << ADSC); // Iniciar conversión
	while (ADCSRA & (1 << ADSC)); // Esperar a que termine conversión
	return ADC;
}

// Pantalla LCD I2C
#define LCD_I2C_ADDR 0x27 // Dirección esclava de
#define LCD_BACKLIGHT 0x08 // Pin retroiluminación activado
#define ENABLE 0x04        // Bit EN
#define RS_PIN 0x01        // Bit RS

// Inicialización del módulo de hardware I2C/TWI
void TWI_init(void) {
	TWSR = 0x00; // Prescaler = 1
	TWBR = 72;   // Frecuencia SCL = 100 kHz a F_CPU = 16 MHz
}

void TWI_start(void) {
	TWCR = (1 << TWINT) | (1 << TWSTA) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT)));
}

void TWI_stop(void) {
	TWCR = (1 << TWINT) | (1 << TWSTO) | (1 << TWEN);
}

void TWI_write(uint8_t data) {
	TWDR = data;
	TWCR = (1 << TWINT) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT)));
}

// Envía byte de datos al expansor de puertos PCF8574
void LCD_I2C_write_pcf(uint8_t data) {
	TWI_start();
	TWI_write((LCD_I2C_ADDR << 1)); // Dirección I2C + bit de Escritura (0)
	TWI_write(data | LCD_BACKLIGHT);
	TWI_stop();
}

// Pulso en EN de LCD, procesa datos
void LCD_I2C_pulse(uint8_t data) {
	LCD_I2C_write_pcf(data | ENABLE);
	_delay_us(1);
	LCD_I2C_write_pcf(data & ~ENABLE);
	_delay_us(50);
}

// Envía byte dividie en 2 (4 bits cada uno)
void LCD_I2C_send(uint8_t value, uint8_t mode) {
	uint8_t highnib = value & 0xF0;
	uint8_t lownib = (value << 4) & 0xF0;
	
	LCD_I2C_write_pcf(highnib | mode);
	LCD_I2C_pulse(highnib | mode);
	
	LCD_I2C_write_pcf(lownib | mode);
	LCD_I2C_pulse(lownib | mode);
}

// Envío comando a LCD I2C (RS = 0)
void LCD_command(uint8_t cmd) {
	LCD_I2C_send(cmd, 0);
	_delay_ms(2);
}

// Envío carácter texto a LCD I2C (RS = 1)
void LCD_char(uint8_t data) {
	LCD_I2C_send(data, RS_PIN);
	_delay_ms(2);
}

// Inicialización  pantalla LCD sobre I2C (modo 4 bits)
void LCD_init(void) {
	TWI_init();
	_delay_ms(50);
	
	// Secuencia de reinicio en 4 bits para el PCF8574
	LCD_I2C_write_pcf(0x30);
	LCD_I2C_pulse(0x30);
	_delay_ms(5);
	
	LCD_I2C_pulse(0x30);
	_delay_us(150);
	
	LCD_I2C_pulse(0x30);
	LCD_I2C_pulse(0x20); // Fijar a modo de comunicación de 4 bits

	LCD_command(0x28); // 2 líneas, formato caracteres 5x8
	LCD_command(0x0C); // Display encendido, sin cursor
	LCD_command(0x01); // Limpiar pantalla
	_delay_ms(5);
}

void LCD_string(const char* str) {
	while (*str) LCD_char(*str++);
}

void LCD_set_cursor(uint8_t row, uint8_t col) {
	uint8_t pos = (row == 0) ? (0x80 + col) : (0xC0 + col);
	LCD_command(pos);
}


// Patrón referencia
typedef struct {
	const char* nombre;
	uint16_t r, g, b; // Valores ADC calibración
	uint16_t m_r, m_g, m_b; // Tolerancia individual
	uint8_t angulo_servo;  // Grados servomotor
	uint8_t led_r, led_g, led_b; // Color matriz WS2812
} ColorRef;

// Tabla patron calibración, tolerancia independiente por color
ColorRef patrones[4] = { //    Valores          Toleracncia    Mov. Servo      Colores Matriz
	// Nombre        ADC_R   ADC_G   ADC_B     M_R   M_G   M_B    Servo   WS2812_R  WS2812_G  WS2812_B
	{"Amarillo",     940,    879,    806,       30,   30,   30,       0,      255,      255,        0},
	{"Verde",        847,    845,    758,       30,   30,   30,      30,        0,      255,        0},
	{"Celeste",      871,    854,    856,       30,   30,   30,      60,        0,      225,      255},
	{"Fucsia",       917,    809,    855,       30,   30,   30,      90,      100,        0,      100}
};

// Algoritmo clasificación, tolerancia individual.
int8_t clasificar_color(uint16_t r, uint16_t g, uint16_t b) {
	for (uint8_t i = 0; i < 4; i++) {
		// Cálculo límite inferior y superior para R
		uint16_t r_min = (patrones[i].r > patrones[i].m_r) ? (patrones[i].r - patrones[i].m_r) : 0;
		uint16_t r_max = patrones[i].r + patrones[i].m_r;

		// Cálculo límite inferior y superior para G
		uint16_t g_min = (patrones[i].g > patrones[i].m_g) ? (patrones[i].g - patrones[i].m_g) : 0;
		uint16_t g_max = patrones[i].g + patrones[i].m_g;

		// Cálculo límite inferior y superior para B
		uint16_t b_min = (patrones[i].b > patrones[i].m_b) ? (patrones[i].b - patrones[i].m_b) : 0;
		uint16_t b_max = patrones[i].b + patrones[i].m_b;

		// Comprobar si lecturas caen dentro de margen patrón
		if ((r >= r_min && r <= r_max) &&
		(g >= g_min && g <= g_max) &&
		(b >= b_min && b <= b_max)) {
			return i; // Color identificado en i
		}
	}
	return -1; // Color no identificado
}


// Main Matriz

int main(void) {
	// Inicializar salidas y periféricos
	WS2812_DDR |= (1 << WS2812_PIN);
	ws2812_clear(); // Limpiar matriz al arranque
	
	UART0_init(103); // 9600 Baudios 16 MHz
	ADC_init();
	Servo_init();
	LCD_init();  // Inicializar LCD I2C

	// Mensaje bienvenida
	LCD_set_cursor(0, 0);
	LCD_string("Clasificador de");
	LCD_set_cursor(1, 0);
	LCD_string("Colorines UwU");
	_delay_ms(1500);
	LCD_command(0x01); // Limpiar pantalla

	char buffer[32];
	uint16_t lectura_R, lectura_G, lectura_B;
	int8_t ultimo_idx = -2; // Seguimiento del estado anterior del servo

	while (1) {
		//Evakuación y muestreo (LEDs 0, 1 y 2)
		
		// Color Rojo y lectura de reflectancia
		ws2812_set_eval_leds(255, 0, 0);
		_delay_ms(150);
		lectura_R = ADC_read(0);

		// Color Verde y lectura de reflectancia
		ws2812_set_eval_leds(0, 255, 0);
		_delay_ms(150);
		lectura_G = ADC_read(0);

		// Color Azul y lectura de reflectancia
		ws2812_set_eval_leds(0, 0, 255);
		_delay_ms(150);
		lectura_B = ADC_read(0);

		//  Clasificación
		int8_t idx = clasificar_color(lectura_R, lectura_G, lectura_B);

		if (idx != -1) {
			// Se reconoce  color y cambió posición:
			if (idx != ultimo_idx) {
				Servo_set_angle(patrones[idx].angulo_servo); // Mover servo (0°, 30°, 60° o 90°)
				ultimo_idx = idx;
			}

			// Resultado matriz ( 8 al 63)
			ws2812_set_display_leds(patrones[idx].led_r, patrones[idx].led_g, patrones[idx].led_b);

			// Enviar datos por  UART
			sprintf(buffer, "RGB:(%u,%u,%u)->%s\r\n", lectura_R, lectura_G, lectura_B, patrones[idx].nombre);
			UART0_send_string(buffer);

			//  Mostrar resultado en pantalla
			LCD_set_cursor(0, 0);
			sprintf(buffer, "Color: %-9s", patrones[idx].nombre);
			LCD_string(buffer);

			LCD_set_cursor(1, 0);
			sprintf(buffer, "R:%-3u G:%-3u B:%-3u", lectura_R, lectura_G, lectura_B);
			LCD_string(buffer);
			} else {
			// Lectura  fuera  margen:
			ws2812_set_display_leds(0, 0, 0); // Apagar área visualización
			
			// Notificar por UART
			sprintf(buffer, "RGB:(%u,%u,%u)->Desconocido\r\n", lectura_R, lectura_G, lectura_B);
			UART0_send_string(buffer);

			// Notificar por Pantalla LCD
			LCD_set_cursor(0, 0);
			LCD_string("Color:Desconocido");
			
			LCD_set_cursor(1, 0);
			sprintf(buffer, "R:%-3u G:%-3u B:%-3u", lectura_R, lectura_G, lectura_B);
			LCD_string(buffer);
		}

		// Espera entre ciclo y ciclo (3s)
		_delay_ms(3000);
	}

	return 0;
}
