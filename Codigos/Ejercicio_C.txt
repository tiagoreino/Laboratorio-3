#define F_CPU 16000000UL // Frecuencia reloj del ATmega2560 (16 MHz)
#include <avr/io.h>
#include <util/delay.h>
#include <avr/interrupt.h>
#include <stdio.h>
#include <stdlib.h>
#include <math.h>


// Configuración de la Matiz

#define WS2812_PIN PH3 // Pin datos matriz (6)
#define WS2812_DDR DDRH  // Registro dirección Puerto H
#define WS2812_PORT PORTH  // Registro salida Puerto H

#define NUM_LEDS 64 // Total leds a usar en matriz 8x8

// Estructura color RGB en orden GRB (formato interno WS2812)
typedef struct {
	uint8_t g;
	uint8_t r;
	uint8_t b;
} cRGB;

cRGB leds[NUM_LEDS]; // Búfer memoria de 64 LEDs

// Envío bit a bit con temporización estricta por ensamblador inline (a 16 MHz)
void ws2812_send_byte(uint8_t b) {
	for (uint8_t i = 0; i < 8; i++) {
		if (b & 0x80) {
			// Bit 1 (Pulso alto 700ns, bajo 600ns)
			WS2812_PORT |= (1 << WS2812_PIN); // se enciende el pin
			asm volatile ("nop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\t"); // Se queda congelado (nanosegundos)
			WS2812_PORT &= ~(1 << WS2812_PIN); // Se apaga el pin
			asm volatile ("nop\n\tnop\n\t"); // Mini espera antes de enviar siguente señal
			} else {
			// Bit 0 ( Pulso alto 350ns, bajo 800ns)
			WS2812_PORT |= (1 << WS2812_PIN);
			asm volatile ("nop\n\tnop\n\t");
			WS2812_PORT &= ~(1 << WS2812_PIN);
			asm volatile ("nop\n\tnop\n\tnop\n\tnop\n\tnop\n\tnop\n\t");
		}
		b <<= 1;
	}
}

// Envía trama completa de búfer hacia matriz, desactiva interrupciones
void ws2812_show(void) {
	cli(); // Desactiva interrupciones para no alterar tiempos
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		ws2812_send_byte(leds[i].g);
		ws2812_send_byte(leds[i].r);
		ws2812_send_byte(leds[i].b);
	}
	sei(); // Reactivar interrupciones
	_delay_us(60); // Senal Reset, fija colores
}

// Apaga 64 LEDs de la matriz
void ws2812_clear(void) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		leds[i].r = 0;
		leds[i].g = 0;
		leds[i].b = 0;
	}
	ws2812_show();
}

// Muestreo
// Enciende LEDs 0, 1 y 2 para iluminación RGB
void ws2812_set_eval_leds(uint8_t r, uint8_t g, uint8_t b) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		if (i < 3) {
			leds[i].r = r;
			leds[i].g = g;
			leds[i].b = b;
			} else { // Si no son los preimeros tres leds s epongan en 0
			leds[i].r = 0;
			leds[i].g = 0;
			leds[i].b = 0;
		}
	}
	ws2812_show(); // se llama para enviar información recibida
}

// Resultado
// Enciende LEDs 8 al 63
void ws2812_set_display_leds(uint8_t r, uint8_t g, uint8_t b) {
	for (uint16_t i = 0; i < NUM_LEDS; i++) {
		if (i < 8) {
			// LEDs 0 al 7 apagada
			leds[i].r = 0;
			leds[i].g = 0;
			leds[i].b = 0;
			} else {
			// LEDs 8 al 63 encendidos con color clasificado
			leds[i].r = r;
			leds[i].g = g;
			leds[i].b = b;
		}
	}
	ws2812_show(); // Refrescaa la matriz
}
