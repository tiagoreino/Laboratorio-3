int main(void)
{
	// Entradas con pull-up: PD2-PD7 y PC0-PC1
	DDRD &= ~0xFC;  PORTD |= 0xFC;
	DDRC &= ~0x03;  PORTC |= 0x03;

	// Salidas de buzzers
	DDRB |= (1 << PB2) | (1 << PB3);

	// Timer 1: CTC
	TCCR1A = 0; // Apagado inicialmente
	TCCR1B = (1 << WGM12);

	// Timer 2: CTC
	TCCR2A = (1 << WGM21);
	TCCR2B = 0; // Apagado inicialmente

	UART_init(103);
	UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1: Primavera de Vivaldi\r\n2 - Cancion 2: Mision Imposible\r\n3 - Cancion 3: Chacarera del olvido\r\n");

	while (1)
	{
		// Seleccion de modos
		
		char recibido = UART_recibir();
		switch (recibido) {

			case '0':
			piano();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1: Primavera de Vivaldi\r\n2 - Cancion 2: Mision Imposible\r\n3 - Cancion 3: Chacarera del olvido\r\n");
			break;

			case '1':
			cancion1();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1: Primavera de Vivaldi\r\n2 - Cancion 2: Mision Imposible\r\n3 - Cancion 3: Chacarera del olvido\r\n");
			break;

			case '2':
			cancion2();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1: Primavera de Vivaldi\r\n2 - Cancion 2: Mision Imposible\r\n3 - Cancion 3: Chacarera del olvido\r\n");
			break;

			case '3':
			cancion3();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1: Primavera de Vivaldi\r\n2 - Cancion 2: Mision Imposible\r\n3 - Cancion 3: Chacarera del olvido\r\n");

// MODO PIANO
void piano(void) {
	static const uint16_t notas[8] = {262, 294, 330, 349, 392, 440, 494, 523}; //NOTAS

	UART_mandarPalabras("Modo Piano Activo. Presione botones. Envie 'q' por UART para salir...\r\n");

	while (1) {

		// Comprobacion de salida al menu
		if (UCSR0A & (1 << RXC0)) {
			if (UDR0 == 'q') break;
		}

		// Leemos botones
		uint8_t botones = ((~PIND) >> 2) & 0x3F;
		botones |= ((~PINC) & 0x03) << 6;

		// Buscamos la nota mas grave y la mas aguda entre las apretadas
		uint16_t grave = 0, agudo = 0;
		for (uint8_t i = 0; i < 8; i++) {
			if (botones & (1 << i)) {
				if (grave == 0) grave = notas[i]; // primera apretada = la mas grave
				agudo = notas[i];                 // la ultima que quede = la mas aguda
			}
		}

		sonar_frecuencia_buzzer1(agudo); // melodia
		sonar_frecuencia_buzzer2(grave); // bajo

		_delay_ms(20); // Delay debouncer basico
	}

	// Aseguramos no dejar buzzers sonando
	sonar_frecuencia_buzzer1(0);
	sonar_frecuencia_buzzer2(0);
	UART_mandarPalabras("Volviendo al menu principal...\r\n");
}

// MOTOR DE REPRODUCCION (sirve para todas las canciones)
static const Nota *bajo_p;     // tabla de bajo en uso
static uint16_t    bajo_n;     // cantidad de notas del bajo
static uint16_t    bajo_i;     // indice de la nota actual del bajo
static uint16_t    bajo_ms;    // ms que le quedan a esa nota

// Se llama una vez por cada ms de espera
static void bajo_tick(void) {
	if (bajo_ms == 0) {
		if (bajo_i < bajo_n) {
			Nota b;
			memcpy_P(&b, &bajo_p[bajo_i++], sizeof(Nota));
			sonar_frecuencia_buzzer2(b.hz);
			bajo_ms = (uint16_t)b.dur * UNIDAD_MS;
			} else {
			sonar_frecuencia_buzzer2(0);   // se termino el bajo
			return;
		}
	}
	bajo_ms--;
}

// Espera ms milisegundos. Devuelve 1 si llego una 'q' por UART.
static uint8_t esperar_ms(uint16_t ms) {
	while (ms--) {
		if ((UCSR0A & (1 << RXC0)) && UDR0 == 'q') return 1;
		bajo_tick();
		_delay_ms(1);
	}
	return 0;
}

static void reproducir(const Nota *mel, uint16_t n_mel, const Nota *baj, uint16_t n_baj) {
	bajo_p = baj;  bajo_n = n_baj;
	bajo_i = 0;    bajo_ms = 0;

	for (uint16_t i = 0; i < n_mel; i++) {
		Nota n;
		memcpy_P(&n, &mel[i], sizeof(Nota));

		uint16_t total = (uint16_t)n.dur * UNIDAD_MS;
		uint16_t gap   = (n.dur >= 2) ? 12 : 0;   // silencio corto entre notas repetidas

		sonar_frecuencia_buzzer1(n.hz);
		if (esperar_ms(total - gap)) goto salir;

		if (gap) {
			sonar_frecuencia_buzzer1(0);
			if (esperar_ms(gap)) goto salir;
		}
	}

	salir:
	sonar_frecuencia_buzzer1(0);
	sonar_frecuencia_buzzer2(0);
	UART_mandarPalabras("Volviendo al menu principal...\r\n");
}

	reproducir(tema, sizeof(tema) / sizeof(tema[0]),

}

// BUZZER 1 (Pin PB2): Usa Timer 1 (16 bits) con prescaler fijo de 64
void sonar_frecuencia_buzzer1(uint16_t hz) {
	static uint16_t actual = 0;
	if (hz == actual) return; // solo actuar si cambio la nota
	actual = hz;

	if (hz == 0) {
		TCCR1B &= ~((1 << CS12) | (1 << CS11) | (1 << CS10));
		TCCR1A &= ~(1 << COM1B0);
		PORTB &= ~(1 << PB2);
		return;
	}

	uint16_t ocr = (F_CPU / (2UL * 64UL * hz)) - 1;
	OCR1A = ocr;
	OCR1B = ocr;
	TCNT1 = 0;
	TCCR1A |= (1 << COM1B0);
	TCCR1B |= (1 << CS11) | (1 << CS10); // prescaler 64
}

// BUZZER 2 (Pin PB3): Usa Timer 2 (8 bits) con PRESCALER DINAMICO
void sonar_frecuencia_buzzer2(uint16_t hz) {
	static uint16_t actual = 0;
	if (hz == actual) return; // solo actuar si cambio la nota
	actual = hz;

	if (hz == 0) {
		TCCR2B = 0;
		TCCR2A &= ~(1 << COM2A0);
		PORTB &= ~(1 << PB3);
		return;
	}

	uint32_t ocr_val = (F_CPU / (2UL * 64UL * hz)) - 1;

	// Intentamos primero con prescaler de 64 (Ideal para notas agudas > 490 Hz)
	if (ocr_val <= 255) {
		OCR2A = (uint8_t)ocr_val;
		TCCR2B = (1 << CS22);                             // prescaler 64
	}
	else {
		// Si desborda, intentamos con prescaler de 256 (Notas medias: ~123 Hz a 490 Hz)
		ocr_val = (F_CPU / (2UL * 256UL * hz)) - 1;
		if (ocr_val <= 255) {
			OCR2A = (uint8_t)ocr_val;
			TCCR2B = (1 << CS22) | (1 << CS21);           // prescaler 256
		}
		else {
			// Si sigue desbordando, usamos prescaler de 1024 (Notas muy graves < 123 Hz)
			ocr_val = (F_CPU / (2UL * 1024UL * hz)) - 1;
			if (ocr_val > 255) ocr_val = 255;
			OCR2A = (uint8_t)ocr_val;
			TCCR2B = (1 << CS22) | (1 << CS21) | (1 << CS20); // prescaler 1024
		}
	}

	TCNT2 = 0;
	TCCR2A |= (1 << COM2A0);              // reconecta pin
}

// Inicializacion de comunicacion UART
void UART_init(unsigned int ubrr) {
	UBRR0H = (unsigned char)(ubrr >> 8);
	UBRR0L = (unsigned char)ubrr;
	UCSR0B = (1 << RXEN0) | (1 << TXEN0);
	UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
}

// Enviar caracter por UART
void UART_mandarLetra(char data) {
	while (!(UCSR0A & (1 << UDRE0)));
	UDR0 = data;
}

// Recibir Caracter por UART
char UART_recibir(void) {
	while (!(UCSR0A & (1 << RXC0)));
	return UDR0;
}

// Enviar cadena de caracteres mediante UART
void UART_mandarPalabras(const char *str) {
	while (*str) {
		UART_mandarLetra(*str++);
	}
}
