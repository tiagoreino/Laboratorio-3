#define F_CPU 16000000UL

#include <avr/io.h>

// Notas (Hz) - Mi mayor
#define B4   494
#define D5s  622
#define E5   659
#define F5s  740
#define G5s  831
#define A5   880
#define B5   988
#define C6s  1109
#define SIL  0

typedef struct { uint16_t hz; uint8_t dur; } Nota;   // duracion en 32avos

// LLAMADO A FUNCIONES
void UART_init(unsigned int ubrr);
void UART_mandarLetra(char data);
char UART_recibir(void);
void UART_mandarPalabras(const char *str);

// Buzzers
void sonar_frecuencia_buzzer1(uint16_t hz); // Controla PB2 (Timer 1)
void sonar_frecuencia_buzzer2(uint16_t hz); // Controla PB3 (Timer 2)

// Modos
void piano(void);
void cancion1(void);
void cancion2(void);
void cancion3(void);

int main(void)
{
	// Entradas con pull-up: PD2-PD7 y PC0-PC1
	DDRD &= ~0xFC;  PORTD |= 0xFC;
	DDRC &= ~0x03;  PORTC |= 0x03;

	// Salidas de buzzers
	DDRB |= (1 << PB2) | (1 << PB3);

	// Timer 1: CTC
	TCCR1A = 0; // Apagado inicialmente
	TCCR1B = (1 << WGM12);

	// Timer 2: CTC
	TCCR2A = (1 << WGM21);
	TCCR2B = 0; // Apagado inicialmente

	UART_init(103);
	UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1\r\n2 - Cancion 2\r\n3 - Cancion 3\r\n");

	while (1)
	{
		// Seleccion de modos
		
		char recibido = UART_recibir();
		switch (recibido) {

			case '0':
			piano();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1\r\n2 - Cancion 2\r\n3 - Cancion 3\r\n");
			break;

			case '1':
			cancion1();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1\r\n2 - Cancion 2\r\n3 - Cancion 3\r\n");
			break;

			case '2':
			cancion2();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1\r\n2 - Cancion 2\r\n3 - Cancion 3\r\n");
			break;

			case '3':
			cancion3();
			UART_mandarPalabras("Selecciona el modo de operacion!\r\n0 - Piano\r\n1 - Cancion 1\r\n2 - Cancion 2\r\n3 - Cancion 3\r\n");
			break;

			case '\r':
			case '\n':
			break; // ignorar fin de linea

			default:
			UART_mandarPalabras("Seleccion no valida !!! Vamo arriba Jesus, deja de probar cosas\r\n");
			break;
		}
	}
	return 0;
}


// MODO PIANO
void piano(void) {
	static const uint16_t notas[8] = {262, 294, 330, 349, 392, 440, 494, 523}; //NOTAS

	UART_mandarPalabras("Modo Piano Activo. Presione botones. Envie 'q' por UART para salir...\r\n");

	while (1) {

		// Comprobacion de salida al menu
		if (UCSR0A & (1 << RXC0)) {
			if (UDR0 == 'q') break;
		}

		// Leemos botones
		uint8_t botones = ((~PIND) >> 2) & 0x3F;
		botones |= ((~PINC) & 0x03) << 6;

		// Buscamos la nota mas grave y la mas aguda entre las apretadas
		uint16_t grave = 0, agudo = 0;
		for (uint8_t i = 0; i < 8; i++) {
			if (botones & (1 << i)) {
				if (grave == 0) grave = notas[i]; // primera apretada = la mas grave
				agudo = notas[i];                 // la ultima que quede = la mas aguda
			}
		}

		sonar_frecuencia_buzzer1(agudo); // melodia
		sonar_frecuencia_buzzer2(grave); // bajo

		_delay_ms(20); // Delay debouncer basico
	}

	// Aseguramos no dejar buzzers sonando
	sonar_frecuencia_buzzer1(0);
	sonar_frecuencia_buzzer2(0);
	UART_mandarPalabras("Volviendo al menu principal...\r\n");
}


// Modo canciones
/* CANCION 1: "La Primavera" (Las Cuatro Estaciones) - Antonio Vivaldi
 * Seccion A (ritornello), solo violin -> buzzer 1.
 * Duraciones en 32avos: 16avo=2, corchea=4, negra=8, negra con puntillo=12 */

#define COMPAS_1 {G5s,4},{G5s,4},{G5s,4},{F5s,2},{E5,2},{B5,12},{B5,2},{A5,2}
#define COMPAS_3 {G5s,4},{A5,2},{B5,2},{A5,4},{G5s,4},{F5s,4},{D5s,4},{B4,4},{E5,4}
#define COMPAS_7 {B5,4},{A5,2},{G5s,2},{A5,4},{B5,4},{C6s,4},{B5,8},{E5,4}

static const Nota tema[] PROGMEM = {
	{E5,4},                                   // anacrusa
	COMPAS_1,                                 // c.1
	COMPAS_1,                                 // c.2
	COMPAS_3,                                 // c.3
	COMPAS_1,                                 // c.4
	COMPAS_1,                                 // c.5
	{G5s,4},{A5,2},{B5,2},{A5,4},{G5s,4},{F5s,8},{SIL,4},{E5,4},   // c.6
	COMPAS_7,                                 // c.7  (piano/forte)
	COMPAS_7,                                 // c.8
	{C6s,4},{B5,8},{A5,4},{G5s,4},{F5s,2},{E5,2},                  // c.9
	{F5s,1},{G5s,1},{F5s,1},{G5s,1},{F5s,1},{G5s,1},{F5s,1},{G5s,1}, // trino
	{F5s,8},{SIL,4},{E5,4},{B5,4},{A5,2},{G5s,2},{A5,4},{B5,4},    // c.10
	{C6s,4},{B5,8},{E5,4},{B5,4},{A5,2},{G5s,2},{A5,4},{B5,4},     // c.11
	{C6s,4},{B5,8},{E5,4},{C6s,4},{B5,8},{A5,4},                   // c.12
	{E5,16}                                   // cierre
};

#define UNIDAD_MS 60   // duracion de un 32avo (baja el numero para ir mas rapido)

// Espera ms milisegundos. Devuelve 1 si llego una 'q' por UART.
static uint8_t esperar_ms(uint16_t ms) {
	while (ms--) {
		if ((UCSR0A & (1 << RXC0)) && UDR0 == 'q') return 1;
		_delay_ms(1);
	}
	return 0;
}

void cancion1(void) {
	UART_mandarPalabras("Cancion 1: La Primavera (Vivaldi). Envie 'q' para salir...\r\n");

	for (uint16_t i = 0; i < sizeof(tema) / sizeof(tema[0]); i++) {
		Nota n;
		memcpy_P(&n, &tema[i], sizeof(Nota));

		uint16_t total = (uint16_t)n.dur * UNIDAD_MS;
		uint16_t gap   = (n.dur >= 3) ? 12 : 0;   // silencio corto para separar notas repetidas

		sonar_frecuencia_buzzer1(n.hz);
		if (esperar_ms(total - gap)) goto salir;

		if (gap) {
			sonar_frecuencia_buzzer1(0);
			if (esperar_ms(gap)) goto salir;
		}
	}

salir:
	sonar_frecuencia_buzzer1(0);
	sonar_frecuencia_buzzer2(0);
	UART_mandarPalabras("Volviendo al menu principal...\r\n");
}
void cancion2(void) {}
void cancion3(void) {}


// BUZZER 1 (Pin PB2): Usa Timer 1 (16 bits) con prescaler fijo de 64
void sonar_frecuencia_buzzer1(uint16_t hz) {
	static uint16_t actual = 0;
	if (hz == actual) return; // solo actuar si cambio la nota
	actual = hz;

	if (hz == 0) {
		TCCR1B &= ~((1 << CS12) | (1 << CS11) | (1 << CS10));
		TCCR1A &= ~(1 << COM1B0);
		PORTB &= ~(1 << PB2);
		return;
	}

	uint16_t ocr = (F_CPU / (2UL * 64UL * hz)) - 1;
	OCR1A = ocr;
	OCR1B = ocr;
	TCNT1 = 0;
	TCCR1A |= (1 << COM1B0);
	TCCR1B |= (1 << CS11) | (1 << CS10); // prescaler 64
}


// BUZZER 2 (Pin PB3): Usa Timer 2 (8 bits) con PRESCALER DINAMICO
void sonar_frecuencia_buzzer2(uint16_t hz) {
	static uint16_t actual = 0;
	if (hz == actual) return; // solo actuar si cambio la nota
	actual = hz;

	if (hz == 0) {
		TCCR2B = 0;
		TCCR2A &= ~(1 << COM2A0);
		PORTB &= ~(1 << PB3);
		return;
	}

	uint32_t ocr_val = (F_CPU / (2UL * 64UL * hz)) - 1;

	// Intentamos primero con prescaler de 64 (Ideal para notas agudas > 490 Hz)
	if (ocr_val <= 255) {
		OCR2A = (uint8_t)ocr_val;
		TCCR2B = (1 << CS22);                             // prescaler 64
	}
	else {
		// Si desborda, intentamos con prescaler de 256 (Notas medias: ~123 Hz a 490 Hz)
		ocr_val = (F_CPU / (2UL * 256UL * hz)) - 1;
		if (ocr_val <= 255) {
			OCR2A = (uint8_t)ocr_val;
			TCCR2B = (1 << CS22) | (1 << CS21);           // prescaler 256
		}
		else {
			// Si sigue desbordando, usamos prescaler de 1024 (Notas muy graves < 123 Hz)
			ocr_val = (F_CPU / (2UL * 1024UL * hz)) - 1;
			if (ocr_val > 255) ocr_val = 255;
			OCR2A = (uint8_t)ocr_val;
			TCCR2B = (1 << CS22) | (1 << CS21) | (1 << CS20); // prescaler 1024
		}
	}

	TCNT2 = 0;
	TCCR2A |= (1 << COM2A0);              // reconecta pin
}


// Inicializacion de comunicacion UART
void UART_init(unsigned int ubrr) {
	UBRR0H = (unsigned char)(ubrr >> 8);
	UBRR0L = (unsigned char)ubrr;
	UCSR0B = (1 << RXEN0) | (1 << TXEN0);
	UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
}

// Enviar caracter por UART
void UART_mandarLetra(char data) {
	while (!(UCSR0A & (1 << UDRE0)));
	UDR0 = data;
}

// Recibir Caracter por UART
char UART_recibir(void) {
	while (!(UCSR0A & (1 << RXC0)));
	return UDR0;
}

// Enviar cadena de caracteres mediante UART
void UART_mandarPalabras(const char *str) {
	while (*str) {
		UART_mandarLetra(*str++);
	}
}
