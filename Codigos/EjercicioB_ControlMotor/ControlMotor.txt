#define F_CPU 16000000UL

#include <avr/io.h>
#include <util/delay.h>
#include <stdlib.h>
#include <stdint.h>

//===========================================================
// DEFINICIONES
//===========================================================

#define BAUD 9600
#define UBRR_VALOR ((F_CPU / 16 / BAUD) - 1)

#define CANAL_POT 0      // ADC0 = PC0
#define CANAL_LDR 1      // ADC1 = PC1

//===========================================================
// PINES DEL MOTOR
//===========================================================
//
// PD4       -> IN1 / IN A
// PD5       -> IN2 / IN B
// PD6/OC0A  -> PWM / ENABLE
//
//===========================================================

#define MOTOR_IN1 PD4
#define MOTOR_IN2 PD5
#define MOTOR_PWM PD6


//===========================================================
// VARIABLE GLOBAL DEL SETPOINT
//===========================================================

uint16_t setpoint = 50;


//===========================================================
// FUNCION PARA ESCALAR ADC DE 0-1023 A 0-100 %
//===========================================================

uint16_t escalar_porcentaje(uint16_t valor)
{
	return ((uint32_t)valor * 100) / 1023;
}


//===========================================================
// INICIALIZAR ADC
//===========================================================

void ADC_init(void)
{
	// Referencia AVcc
	ADMUX = (1 << REFS0);

	// Habilitar ADC
	// Prescaler = 128
	//
	// 16 MHz / 128 = 125 kHz

	ADCSRA = (1 << ADEN)  |
	(1 << ADPS2) |
	(1 << ADPS1) |
	(1 << ADPS0);
}


//===========================================================
// LEER ADC
//===========================================================

uint16_t ADC_leer(uint8_t canal)
{
	// Seleccionar referencia AVcc y canal ADC
	ADMUX = (1 << REFS0) | (canal & 0x0F);

	// Iniciar conversión
	ADCSRA |= (1 << ADSC);

	// Esperar hasta que termine
	while (ADCSRA & (1 << ADSC));

	return ADC;
}


//===========================================================
// INICIALIZAR UART
//===========================================================

void UART_init(void)
{
	UBRR0H = (uint8_t)(UBRR_VALOR >> 8);
	UBRR0L = (uint8_t)UBRR_VALOR;

	// UART modo normal
	UCSR0A = 0x00;

	// Habilitar receptor y transmisor
	UCSR0B = (1 << RXEN0) |
	(1 << TXEN0);

	// 8 bits
	// 1 bit de parada
	// Sin paridad

	UCSR0C = (1 << UCSZ01) |
	(1 << UCSZ00);
}


//===========================================================
// ENVIAR CARACTER
//===========================================================

void UART_enviar_char(char dato)
{
	while (!(UCSR0A & (1 << UDRE0)));

	UDR0 = dato;
}


//===========================================================
// ENVIAR TEXTO
//===========================================================

void UART_enviar_texto(const char *texto)
{
	while (*texto)
	{
		UART_enviar_char(*texto);

		texto++;
	}
}


//===========================================================
// ENVIAR NUMERO
//===========================================================

void UART_enviar_numero(uint16_t numero)
{
	char buffer[10];

	itoa(numero, buffer, 10);

	UART_enviar_texto(buffer);
}


//===========================================================
// ENVIAR NUMERO CON SIGNO
//===========================================================

void UART_enviar_numero_signo(int16_t numero)
{
	char buffer[10];

	itoa(numero, buffer, 10);

	UART_enviar_texto(buffer);
}


//===========================================================
// VERIFICAR SI HAY DATO UART
//===========================================================

uint8_t UART_dato_disponible(void)
{
	return (UCSR0A & (1 << RXC0));
}


//===========================================================
// RECIBIR CARACTER
//===========================================================

char UART_recibir_char(void)
{
	return UDR0;
}


//===========================================================
// REVISAR NUEVO SETPOINT
//===========================================================

void UART_revisar_setpoint(void)
{
	char buffer_rx[6];

	uint8_t indice = 0;

	char caracter;

	uint16_t nuevo_setpoint;


	//-------------------------------------------------------
	// SI NO LLEGO NADA, SALIR
	//-------------------------------------------------------

	if (!UART_dato_disponible())
	{
		return;
	}


	//=======================================================
	// RECIBIR TODOS LOS CARACTERES ENVIADOS CON SEND
	//=======================================================

	while (1)
	{
		if (UART_dato_disponible())
		{
			caracter = UART_recibir_char();


			//------------------------------------------------
			// GUARDAR SOLO DIGITOS
			//------------------------------------------------

			if (caracter >= '0' && caracter <= '9')
			{
				// Máximo 3 dígitos: 0 - 100

				if (indice < 3)
				{
					buffer_rx[indice] = caracter;

					indice++;
				}
			}


			//------------------------------------------------
			// ESPERAR POSIBLE SIGUIENTE CARACTER
			//------------------------------------------------

			_delay_ms(2);
		}
		else
		{
			break;
		}
	}


	//=======================================================
	// PROCESAR SETPOINT
	//=======================================================

	if (indice > 0)
	{
		buffer_rx[indice] = '\0';

		nuevo_setpoint = atoi(buffer_rx);


		if (nuevo_setpoint <= 100)
		{
			setpoint = nuevo_setpoint;
		}
	}
}


//===========================================================
// INICIALIZAR MOTOR Y PWM
//===========================================================
//
// PD4 -> Dirección IN1
// PD5 -> Dirección IN2
// PD6 -> PWM OC0A
//
//===========================================================

void MOTOR_init(void)
{
	//=======================================================
	// CONFIGURAR PINES COMO SALIDAS
	//=======================================================

	DDRD |= (1 << MOTOR_IN1) |
	(1 << MOTOR_IN2) |
	(1 << MOTOR_PWM);


	//=======================================================
	// ESTADO INICIAL
	//=======================================================
	//
	// IN1 = 0
	// IN2 = 0
	// PWM = 0
	//
	// Motor detenido
	//
	//=======================================================

	PORTD &= ~(1 << MOTOR_IN1);
	PORTD &= ~(1 << MOTOR_IN2);


	//=======================================================
	// CONFIGURAR TIMER0 EN FAST PWM
	//=======================================================
	//
	// OC0A = PD6
	//
	// Fast PWM 8 bits
	// No invertido
	//
	// WGM01 = 1
	// WGM00 = 1
	//
	// COM0A1 = 1
	//
	//=======================================================

	TCCR0A = (1 << COM0A1) |
	(1 << WGM01)  |
	(1 << WGM00);


	//=======================================================
	// PRESCALER = 64
	//=======================================================
	//
	// Frecuencia PWM:
	//
	// 16 MHz / (64 * 256)
	//
	// aproximadamente 976 Hz
	//
	//=======================================================

	TCCR0B = (1 << CS01) |
	(1 << CS00);


	// PWM inicialmente en 0
	OCR0A = 0;
}


//===========================================================
// CONTROL PROPORCIONAL DEL MOTOR
//===========================================================
//
//                  ERROR
//
//              error = SP - LDR
//
//
// ERROR POSITIVO:
//
//      IN1 = 1
//      IN2 = 0
//
//      Giro DERECHA
//
//
// ERROR NEGATIVO:
//
//      IN1 = 0
//      IN2 = 1
//
//      Giro IZQUIERDA
//
//
// ERROR = 0:
//
//      IN1 = 0
//      IN2 = 0
//
//      Motor PARADO
//
//
// MAGNITUD DEL ERROR:
//
//      0 - 100 %
//
//          ?
//
//      PWM 0 - 255
//
//===========================================================

uint8_t controlar_motor(int16_t error, uint16_t valor_pot)
{
	uint16_t magnitud_error;

	uint8_t pwm;


	//====================================================================================
	// ERROR POSITIVO Y POTENCIOMETRO_POSICION SIN LLEGAR AL MAXIMO -> GIRA A LA DERECHA
	//====================================================================================

	if (error > 0 && valor_pot < 95)
	{
		//---------------------------------------------------
		// OBTENER MAGNITUD DEL ERROR
		//---------------------------------------------------

		magnitud_error = error;


		//---------------------------------------------------
		// ESCALAR ERROR 0-100 A PWM 0-255
		//---------------------------------------------------

		pwm = ((uint32_t)magnitud_error * 255) / 100;


		//---------------------------------------------------
		// SENTIDO DERECHA
		//---------------------------------------------------
		//
		// IN1 = 1
		// IN2 = 0
		//
		//---------------------------------------------------

		PORTD |=  (1 << MOTOR_IN1);

		PORTD &= ~(1 << MOTOR_IN2);


		//---------------------------------------------------
		// APLICAR PWM
		//---------------------------------------------------

		OCR0A = pwm;
	}


	//======================================================================================
	// ERROR NEGATIVO Y POTENCIOMETRO_POSICION SIN LLEGAR AL MINIMO -> GIRA A LA IZQUIERDA
	//======================================================================================

	else if (error < 0 && valor_pot > 5)
	{
		//---------------------------------------------------
		// OBTENER VALOR ABSOLUTO DEL ERROR
		//---------------------------------------------------

		magnitud_error = -error;


		//---------------------------------------------------
		// ESCALAR ERROR 0-100 A PWM 0-255
		//---------------------------------------------------

		pwm = ((uint32_t)magnitud_error * 255) / 100;


		//---------------------------------------------------
		// SENTIDO IZQUIERDA
		//---------------------------------------------------
		//
		// IN1 = 0
		// IN2 = 1
		//
		//---------------------------------------------------

		PORTD &= ~(1 << MOTOR_IN1);

		PORTD |=  (1 << MOTOR_IN2);


		//---------------------------------------------------
		// APLICAR PWM
		//---------------------------------------------------

		OCR0A = pwm;
	}


	//=======================================================
	// ERROR = 0
	//=======================================================

	else
	{
		//---------------------------------------------------
		// MOTOR DETENIDO
		//---------------------------------------------------

		PORTD &= ~(1 << MOTOR_IN1);

		PORTD &= ~(1 << MOTOR_IN2);

		OCR0A = 0;

		pwm = 0;
	}


	return pwm;
}


//===========================================================
// PROGRAMA PRINCIPAL
//===========================================================

int main(void)
{
	uint16_t valor_pot;

	uint16_t valor_ldr;

	int16_t error;

	uint8_t pwm_motor;

	uint16_t contador = 0;


	//=======================================================
	// INICIALIZAR PERIFERICOS
	//=======================================================

	ADC_init();

	UART_init();

	MOTOR_init();


	//=======================================================
	// MENSAJE INICIAL
	//=======================================================

	UART_enviar_texto("\r\n");

	UART_enviar_texto("============================\r\n");

	UART_enviar_texto(" CONTROL PROPORCIONAL\r\n");

	UART_enviar_texto("============================\r\n");

	UART_enviar_texto("Ingrese Setpoint (0-100%) usando SEND\r\n");

	UART_enviar_texto("============================\r\n\r\n");


	//=======================================================
	// BUCLE PRINCIPAL
	//=======================================================

	while (1)
	{

		//===================================================
		// 1. REVISAR NUEVO SETPOINT
		//===================================================

		UART_revisar_setpoint();


		//===================================================
		// 2. LEER POTENCIOMETRO
		//===================================================

		valor_pot = ADC_leer(CANAL_POT);

		valor_pot = escalar_porcentaje(valor_pot);


		//===================================================
		// 3. LEER LDR
		//===================================================

		valor_ldr = ADC_leer(CANAL_LDR);

		valor_ldr = escalar_porcentaje(valor_ldr);


		//===================================================
		// 4. CALCULAR ERROR
		//===================================================
		//
		// ERROR = SP - LDR
		//
		// Rango:
		//
		// -100 <= error <= +100
		//
		//===================================================

		error = (int16_t)setpoint - (int16_t)valor_ldr;


		//===================================================
		// 5. CONTROL PROPORCIONAL
		//===================================================

		pwm_motor = controlar_motor(error, valor_pot);


		//===================================================
		// 6. CONTADOR PARA MUESTREO
		//===================================================

		contador++;


		//===================================================
		// 7. MOSTRAR DATOS CADA 1 SEGUNDO
		//===================================================

		if (contador >= 1000)
		{
			contador = 0;


			//------------------------------------------------
			// POTENCIOMETRO
			//------------------------------------------------

			UART_enviar_texto("POT: ");

			UART_enviar_numero(valor_pot);

			UART_enviar_texto("%");


			//------------------------------------------------
			// LDR
			//------------------------------------------------

			UART_enviar_texto("   LDR: ");

			UART_enviar_numero(valor_ldr);

			UART_enviar_texto("%");


			//------------------------------------------------
			// SETPOINT
			//------------------------------------------------

			UART_enviar_texto("   SP: ");

			UART_enviar_numero(setpoint);

			UART_enviar_texto("%");


			//------------------------------------------------
			// ERROR
			//------------------------------------------------

			UART_enviar_texto("   ERROR: ");

			UART_enviar_numero_signo(error);

			UART_enviar_texto("%");


			//------------------------------------------------
			// PWM
			//------------------------------------------------

			UART_enviar_texto("   PWM: ");

			UART_enviar_numero(pwm_motor);


			//------------------------------------------------
			// SENTIDO
			//------------------------------------------------

			UART_enviar_texto("   SENTIDO: ");


			if (error > 0)
			{
				UART_enviar_texto("DERECHA");
			}

			else if (error < 0)
			{
				UART_enviar_texto("IZQUIERDA");
			}

			else
			{
				UART_enviar_texto("PARADO");
			}


			//------------------------------------------------
			// NUEVA LINEA
			//------------------------------------------------

			UART_enviar_texto("\r\n");
		}


		//===================================================
		// BASE DE TIEMPO
		//===================================================

		_delay_ms(1);
	}
}
